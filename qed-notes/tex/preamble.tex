\newcommand{\keybox}[2]{%

  \begin{tikzpicture}
    \node[draw=cAmber,fill=cAmberBg,line width=0.9pt,rounded corners=3pt,
          inner sep=8pt,text width=0.92\linewidth,align=left]
      {\textbf{\color{cAmber}#1}\\[2pt]#2};
  \end{tikzpicture}
  \end{center}}
